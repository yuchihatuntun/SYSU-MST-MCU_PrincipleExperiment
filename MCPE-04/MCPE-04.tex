\section{实验内容与设计}

\subsection{循环程序实验}

\begin{lstlisting}[language={[x86masm]Assembler}]
	CODE    SEGMENT
	ASSUME  CS:CODE          
	CON_A   EQU 25           
	CON_B   EQU 12           
	
	START:  MOV     AX, 0           
	MOV     CX, 5           
	INC_AX: NOP                      
	INC     AX               
	LOOP    INC_AX          ; LOOP会自动递减CX
	JMP     $               
	CODE    ENDS                     
	END     START  
\end{lstlisting}

\subsection{分支程序实验}

\begin{lstlisting}[language={[x86masm]Assembler}]
CODE    SEGMENT
ASSUME  CS:CODE          
CON_A   EQU 25           
CON_B   EQU 12           

START:  MOV     AX, CON_A       
MOV     BX, CON_B       
CMP     AX, BX          
JNC     MO_T            ; AX ≥ BX（无进位），跳至MO_T
JE      EQUA            ; AX = BX（相等），跳至EQUA
JC      LESS            ; AX < BX（有进位），跳至LESS

MO_T:   MOV     CX, 0100H
JMP     $       
EQUA:   MOV     CX, 0200H  
JMP     $     
LESS:   MOV     CX, 0300H               
JMP     $

CODE    ENDS                    
END     START     
\end{lstlisting}

\subsection{内存块移动实验}

\begin{figure}[H]\centering
	\includegraphics[width=1\columnwidth]{MCPE-04 Memory Block Movement Experiment/assets/内存块移动实验-flowchart.png}
	\caption{程序流程图}
\end{figure}

\begin{lstlisting}[language={[x86masm]Assembler}]
CODE    SEGMENT
ASSUME  CS:CODE, DS:CODE, ES:CODE  ; 段定义

START:  
MOV     AX, CS
MOV     DS, AX            ; DS = CS
MOV     ES, AX            ; ES = CS

MOV     SI, 1000H         
MOV     CX, 10            
MOV     AL, 1
INIT_DATA:
MOV     [SI], AL
INC     AL
INC     SI
LOOP    INIT_DATA


MOV     SI, 1000H         
MOV     DI, 1002H         ; 目的首地址
MOV     CX, 10            

CMP     SI, DI            ; 比较SI和DI
JAE     USE_FORWARD       ; CF=0，对应SI >= DI 
; SI < DI  

USE_BACKWARD:
ADD     SI, CX
DEC     SI                ; SI = SI + CX - 1
ADD     DI, CX
DEC     DI                ; DI = DI + CX - 1

STD                       ; DF=1 
JMP     DO_MOVE           

USE_FORWARD:
CLD                       ; DF=0 

DO_MOVE:
REP     MOVSB             
JMP     $                 
CODE    ENDS
END     START
\end{lstlisting}

\subsection{附加实验：语句相等判断分支}

\begin{figure}[H]\centering
	\includegraphics[width=1\columnwidth]{MCPE-04 Memory Block Movement Experiment/assets/语句相等判断分支.png}
	\caption{程序流程图}

\begin{lstlisting}[language={[x86masm]Assembler}]
	CODE    SEGMENT
	ASSUME  CS:CODE, DS:CODE, ES:CODE  
	
	START:  
	MOV     AX, CS
	MOV     DS, AX            
	MOV     ES, AX            
	
	
	MOV     SI, 1000H         ; 字符串1首地址
	MOV     CX, 5             ; 
	MOV     AL, 'A'           ; 第一个字符'A'
	
	INIT_STR1:                        
	MOV     [SI], AL          
	INC     AL                
	INC     SI                
	LOOP    INIT_STR1         
	
	
	MOV     DI, 2000H         ; 字符串2首地址
	MOV     CX, 5             
	MOV     AL, 'B'           ; 第一个字符'B'
	
	INIT_STR2:                        
	MOV     [DI], AL          
	INC     AL                
	INC     DI                
	LOOP    INIT_STR2         
	
	MOV     SI, 1000H         
	MOV     DI, 2000H         
	MOV     CX, 5             
	
	CLD                       ; DF=0，在串操作时自动递增
	
	; - REPE (Repeat While Equal)
	; - CMPSB (Compare Byte Strings)
	REPE    CMPSB             
	
	
	JZ      MATCH_ADDR        ; ZF=1，跳转到MATCH_ADDR
	JNZ     NOMATCH_ADDR      ; ZF=0，跳转到NOMATCH_ADDR
	
	
	MATCH_ADDR:                       
	MOV     AX, 0000H         
	JMP     STOP_PROG         
	
	NOMATCH_ADDR:                     
	MOV     AX, 0FFFFH        
	JMP     STOP_PROG        
	
	STOP_PROG:                        
	JMP     $                 
	
	CODE    ENDS
	END     START             
\end{lstlisting}

\section{实验环节分析与优化}

本次实验通过四个核心环节系统性地掌握了8086汇编语言的核心编程技术。从基础的循环控制到复杂的内存操作，逐步深入理解了汇编语言在底层硬件控制中的重要作用。实验过程中，通过实际操作验证了理论知识的正确性，特别是在条件标志位的判断和内存地址处理方面获得了宝贵经验。

\subsection{循环程序实验的优化空间}

\textbf{当前实现：}实验采用LOOP指令实现简单的计数循环，结构清晰但功能单一。

\textbf{优化方向：}
\begin{itemize}
	\item \textbf{循环结构多样化：}可增加嵌套循环设计，处理更复杂的数据结构
	\item \textbf{循环控制优化：}改用DEC+JNZ组合替代LOOP指令，提高代码兼容性
	\item \textbf{循环边界检测：}增加CX值有效性检查，避免死循环情况
\end{itemize}

\begin{lstlisting}[language={[x86masm]Assembler}]
	CMP CX, 0
	JLE EXIT_LOOP
	INC_AX:
	; 循环体
	DEC CX
	JNZ INC_AX
	EXIT_LOOP:
\end{lstlisting}

\subsection{分支程序实验的改进方案}
\textbf{现有不足：}条件判断相对简单，缺乏错误处理机制。

\textbf{优化建议：}
\begin{itemize}
	\item \textbf{多条件分支：}扩展为多级判断结构，处理更复杂的比较场景
	\item \textbf{标志位保护：}在关键跳转前保存重要标志位状态
	\item \textbf{分支预测优化：}调整代码顺序，将概率更高的分支放在前面
\end{itemize}

\subsection{内存块移动实验的进阶优化}

实验实现了智能的方向判断机制，根据SI和DI的相对位置自动选择复制方向。

\textbf{深度优化：}
\begin{itemize}
	\item \textbf{性能提升：}对于大块数据，可采用MOVSW代替MOVSB，减少指令执行次数
	\item \textbf{边界对齐：}增加地址对齐检查，优化内存访问效率
	\item \textbf{重叠检测增强：}完善重叠区域检测算法，确保数据完整性
\end{itemize}

\begin{lstlisting}[language={[x86masm]Assembler}]
	CMP SI, DI
	JB BACKWARD_COPY
	JE NO_OPERATION  ; 增加相同地址处理
	
	FORWARD_COPY:
	CLD
	SHR CX, 1     ; 字传输优化
	REP MOVSW
	JMP COPY_END
\end{lstlisting}

\subsection{字符串比较实验的扩展可能}

通过REPE CMPSB指令实现了高效的字符串比较功能。

\textbf{功能扩展：}

\begin{itemize}
	\item \textbf{比较结果细化：}不仅判断是否相等，还能返回首个不匹配字符位置
	\item \textbf{大小写不敏感比较：}增加字符规范化处理逻辑
	\item \textbf{长度自适应：}支持可变长度字符串比较
\end{itemize}
